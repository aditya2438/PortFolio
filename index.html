@RestController
@RequestMapping("/api/v1/cart")
public class CartController {

    private final CartService cartService;
    private final KafkaTemplate<String, OrderEvent> kafkaTemplate;
    private final RedisTemplate<String, CartSession> redisTemplate;

    @PostMapping("/{userId}/checkout")
    public ResponseEntity<OrderResponse> checkout(
            @PathVariable Long userId,
            @RequestBody CheckoutRequest req) {
        
        // Distributed Redis lock + 3NF pessimistic inventory check
        Order order = cartService.processWithLock(userId, req);
        
        // Asynchronous Kafka dispatch to Dead-Letter Queue (DLQ) topology
        kafkaTemplate.send("order-dispatch-events", order.getId(), new OrderEvent(order));
        
        return ResponseEntity.ok(OrderResponse.from(order));
    }
}
@Service
public class HaversineService {

    private static final double EARTH_RADIUS_KM = 6371.0;

    /**
     * Location-aware provider discovery for target city clusters:
     * Ujjain (23.17°N, 75.78°E), Indore (22.71°N), Bhopal (23.25°N)
     */
    public double calculateDistance(double lat1, double lon1, 
                                     double lat2, double lon2) {
        double dLat = Math.toRadians(lat2 - lat1);
        double dLon = Math.toRadians(lon2 - lon1);
        
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
                   Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
                   Math.sin(dLon / 2) * Math.sin(dLon / 2);
                   
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return EARTH_RADIUS_KM * c;
    }
}
